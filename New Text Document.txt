  <script>
    const PLAYLIST_URL = "https://iptv-org.github.io/iptv/index.m3u";
    const video = document.querySelector("#video");
    const channelsEl = document.querySelector("#channels");
    const statusEl = document.querySelector("#status");
    const nowPlaying = document.querySelector("#nowPlaying");
    const searchEl = document.querySelector("#search");
    const categoryEl = document.querySelector("#category");

    let channels = [];
    let player = null;

    function parsePlaylist(text) {
      const lines = text.split(/\r?\n/);
      const items = [];
      let info = "";

      for (const raw of lines) {
        const line = raw.trim();
        if (line.startsWith("#EXTINF")) {
          info = line;
        } else if (line && !line.startsWith("#")) {
          const name = info.match(/,(.*)$/)?.[1]?.trim() || "নামহীন চ্যানেল";
          const group = info.match(/group-title="([^"]*)"/)?.[1] || "অন্যান্য";
          items.push({ name, group, url: line });
          info = "";
        }
      }
      return items;
    }

    function setupCategories() {
      const groups = [...new Set(channels.map(c => c.group))].sort();
      categoryEl.replaceChildren(new Option("সব ক্যাটাগরি", ""));
      for (const group of groups) categoryEl.add(new Option(group, group));
    }

    function renderChannels() {
      const query = searchEl.value.trim().toLowerCase();
      const group = categoryEl.value;
      const filtered = channels.filter(c =>
        c.name.toLowerCase().includes(query) &&
        (!group || c.group === group)
      );

      channelsEl.replaceChildren();
      for (const channel of filtered.slice(0, 300)) {
        const button = document.createElement("button");
        button.className = "channel";
        button.innerHTML = "";
        button.append(document.createTextNode(channel.name));
        const small = document.createElement("small");
        small.textContent = channel.group;
        button.append(small);
        button.addEventListener("click", () => playChannel(channel, button));
        channelsEl.append(button);
      }

      statusEl.textContent = `${filtered.length}টি চ্যানেল`;
    }

    function playChannel(channel, button) {
      document.querySelectorAll(".channel").forEach(el => el.classList.remove("active"));
      button.classList.add("active");
      nowPlaying.textContent = `এখন চলছে: ${channel.name}`;

      if (player) {
        player.destroy();
        player = null;
      }

      if (Hls.isSupported()) {
        player = new Hls();
        player.loadSource(channel.url);
        player.attachMedia(video);
        player.on(Hls.Events.MANIFEST_PARSED, () => video.play().catch(() => {}));
        player.on(Hls.Events.ERROR, (_, data) => {
          if (data.fatal) nowPlaying.textContent = `স্ট্রিম চালানো যাচ্ছে না: ${channel.name}`;
        });
      } else if (video.canPlayType("application/vnd.apple.mpegurl")) {
        video.src = channel.url;
        video.play().catch(() => {});
      } else {
        nowPlaying.textContent = "এই ব্রাউজারে এই স্ট্রিম ফরম্যাট সমর্থিত নয়।";
      }
    }

    async function loadPlaylist() {
      try {
        const response = await fetch(PLAYLIST_URL);
        if (!response.ok) throw new Error("Playlist request failed");
        channels = parsePlaylist(await response.text());
        setupCategories();
        renderChannels();
      } catch (error) {
        statusEl.textContent = "প্লেলিস্ট লোড হয়নি—নেটওয়ার্ক বা CORS সমস্যা হতে পারে।";
        console.error(error);
      }
    }

    searchEl.addEventListener("input", renderChannels);
    categoryEl.addEventListener("change", renderChannels);
    loadPlaylist();
  </script>
